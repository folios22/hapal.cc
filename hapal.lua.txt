-- ===== УТИЛИТЫ =====
local function WorldToScreen(pos)
    local screenPos, onScreen = Camera:WorldToScreenPoint(pos)
    return Vector2.new(screenPos.X, screenPos.Y), onScreen and screenPos.Z > 0
end

local function GetRootPart(char)
    local names = {"HumanoidRootPart", "Torso", "UpperTorso", "LowerTorso", "RootPart", "Root"}
    for _, name in ipairs(names) do
        local part = char:FindFirstChild(name)
        if part and part:IsA("BasePart") then
            return part
        end
    end
    for _, child in ipairs(char:GetChildren()) do
        if child:IsA("BasePart") then
            return child
        end
    end
    return nil
end

local function GetPart(char, name)
    local part = char:FindFirstChild(name)
    if part and part:IsA("BasePart") then
        return part.Position
    end
    return nil
end

-- ===== БОКС ОТ НОГ ДО ГОЛОВЫ =====
local function GetBoundingBox2D(char)
    local head = char:FindFirstChild("Head")
    local root = GetRootPart(char)
    if not head or not root then return nil end

    local lowestY = root.Position.Y
    local highestY = head.Position.Y

    for _, part in ipairs(char:GetDescendants()) do
        if part:IsA("BasePart") then
            local pos = part.Position
            if pos.Y < lowestY then lowestY = pos.Y end
            if pos.Y > highestY then highestY = pos.Y end
        end
    end

    local topPos = Vector3.new(head.Position.X, highestY, head.Position.Z)
    local bottomPos = Vector3.new(head.Position.X, lowestY, head.Position.Z)

    local topScreen, topOnScreen = WorldToScreen(topPos)
    local bottomScreen, bottomOnScreen = WorldToScreen(bottomPos)
    if not topOnScreen or not bottomOnScreen then return nil end

    local height = math.abs(topScreen.Y - bottomScreen.Y)
    if height < 10 then height = 10 end

    local width = height * 0.4
    if width < 10 then width = 10 end

    local centerX = (topScreen.X + bottomScreen.X) / 2
    local centerY = (topScreen.Y + bottomScreen.Y) / 2

    return centerX - width/2, centerY - height/2, centerX + width/2, centerY + height/2
end

local function IsTeammate(plr)
    if LocalPlayer.Team == nil or plr.Team == nil then return false end
    return LocalPlayer.Team == plr.Team
end

local function IsAlive(char)
    local humanoid = char:FindFirstChildOfClass("Humanoid")
    if humanoid then
        return humanoid.Health > 0 and humanoid:GetState() ~= Enum.HumanoidStateType.Dead
    end
    return true
end

local function IsVisible(part)
    if settings.Wallbang or settings.IgnoreWalls then return true end
    local origin = Camera.CFrame.Position
    local direction = (part.Position - origin).Unit * (part.Position - origin).Magnitude
    local rayParams = RaycastParams.new()
    rayParams.FilterType = Enum.RaycastFilterType.Exclude
    rayParams.FilterDescendantsInstances = {LocalPlayer.Character, part.Parent}
    local result = workspace:Raycast(origin, direction, rayParams)
    return result == nil
end

local currentTarget = nil

local function IsTargetValid(part)
    if not part or not part.Parent then return false end
    local char = part.Parent
    local plr = Players:GetPlayerFromCharacter(char)
    if not plr or plr == LocalPlayer or IsTeammate(plr) then return false end
    if not IsAlive(char) then return false end
    if not IsVisible(part) then return false end

    local screenPos, onScreen = Camera:WorldToScreenPoint(part.Position)
    if not onScreen or screenPos.Z <= 0 then return false end

    local screenCenter = Vector2.new(Camera.ViewportSize.X / 2, Camera.ViewportSize.Y / 2)
    local dist = (Vector2.new(screenPos.X, screenPos.Y) - screenCenter).Magnitude
    return dist <= settings.FOV
end

local function GetClosestTarget()
    if currentTarget and IsTargetValid(currentTarget) then
        return currentTarget
    end

    local screenCenter = Vector2.new(Camera.ViewportSize.X / 2, Camera.ViewportSize.Y / 2)
    local closest = nil
    local closestDist = settings.FOV

    for plr, _ in pairs(espTable) do
        local char = plr.Character
        if char and IsAlive(char) and not IsTeammate(plr) then
            local part = char:FindFirstChild(settings.TargetPart)
            if not part or not part:IsA("BasePart") then
                part = char:FindFirstChild("HumanoidRootPart") or char:FindFirstChild("Head") or char:FindFirstChild("Torso") or char:FindFirstChild("UpperTorso")
            end
            if not part or not part:IsA("BasePart") then
                part = GetRootPart(char)
            end
            if part and IsVisible(part) then
                local pos, onScreen = WorldToScreen(part.Position)
                if onScreen then
                    local dist = (pos - screenCenter).Magnitude
                    if dist < closestDist then
                        closestDist = dist
                        closest = part
                    end
                end
            end
        end
    end

    currentTarget = closest
    return closest
end

    oldNamecall = hookmetamethod(game, "__namecall", function(self, ...)
        local args = {...}
        local method = getnamecallmethod()

        if settings.SilentAim and settings.SilentMethod ~= "Fake" and (method == "FireServer" or method == "InvokeServer") then
            if self:IsA("RemoteEvent") or self:IsA("RemoteFunction") then
                local target = GetClosestTarget()
                if target then
                    for i, v in ipairs(args) do
                        if typeof(v) == "Vector3" then
                            args[i] = target.Position
                        elseif typeof(v) == "CFrame" then
                            args[i] = CFrame.lookAt(v.Position, target.Position)
                        end
                    end
                    return oldNamecall(self, unpack(args))
                end
            end
        end

        return oldNamecall(self, ...)
    end)
    hookWorks = true
end)

if not ok then
    warn("[SilentAim] hookmetamethod не поддерживается — используется Fake метод")

                    local leftArm = GetPart(char, "Left Arm") or GetPart(char, "LeftHand") or GetPart(char, "LeftUpperArm")
                    local rightArm = GetPart(char, "Right Arm") or GetPart(char, "RightHand") or GetPart(char, "RightUpperArm")
                    local leftLeg = GetPart(char, "Left Leg") or GetPart(char, "LeftFoot") or GetPart(char, "LeftUpperLeg")
                    local rightLeg = GetPart(char, "Right Leg") or GetPart(char, "RightFoot") or GetPart(char, "RightUpperLeg")

                    local points = {
                        Head = head and WorldToScreen(head),
                        Torso = torso and WorldToScreen(torso),
                        ["Left Arm"] = leftArm and WorldToScreen(leftArm),
                        ["Right Arm"] = rightArm and WorldToScreen(rightArm),
                        ["Left Leg"] = leftLeg and WorldToScreen(leftLeg),
                        ["Right Leg"] = rightLeg and WorldToScreen(rightLeg),
                    }

                    for conn, line in pairs(esp.Skeleton) do
                        local a, b = conn:match("(.+)_(.+)")
                        local pa = points[a]
                        local pb = points[b]
                        if pa and pb and pa[2] and pb[2] then
                            line.Visible = true
                            line.From = pa[1]
                            line.To = pb[1]
                        else
                            line.Visible = false
                        end
                    end
                else
                    for _, line in pairs(esp.Skeleton) do line.Visible = false end
                end

                -- HEALTH
                if settings.Health then
                    local humanoid = char:FindFirstChildOfClass("Humanoid")
                    local head = char:FindFirstChild("Head")
                    if humanoid and head then
                        local headPos, onScreen = WorldToScreen(head.Position)
                        if onScreen then
                            local health = humanoid.Health / humanoid.MaxHealth
                            if health < 0 then health = 0 end
                            if health > 1 then health = 1 end
                            local barWidth = 50
                            local barHeight = 5
                            local barX = headPos.X - barWidth / 2
                            local barY = headPos.Y - 30

                            esp.HealthBG.Visible = true
                            esp.HealthBG.Position = Vector2.new(barX - 1, barY - 1)
                            esp.HealthBG.Size = Vector2.new(barWidth + 2, barHeight + 2)

                            esp.HealthBar.Visible = true
                            esp.HealthBar.Position = Vector2.new(barX, barY)
                            esp.HealthBar.Size = Vector2.new(barWidth * health, barHeight)
                            esp.HealthBar.Color = Color3.fromRGB(255 * (1 - health), 255 * health, 0)
                        else
                            esp.HealthBG.Visible = false
                            esp.HealthBar.Visible = false
                        end
                    else
                        esp.HealthBG.Visible = false
                        esp.HealthBar.Visible = false
                    end
                else
                    esp.HealthBG.Visible = false
                    esp.HealthBar.Visible = false
                end

                -- CHAMS
                if settings.Chams then
                    if not esp.Highlight then
                        local highlight = Instance.new("Highlight")
                        highlight.Name = "hapalChams"
                        highlight.FillColor = settings.Color
                        highlight.OutlineColor = Color3.fromRGB(255, 255, 255)
                        highlight.FillTransparency = 0.5
                        highlight.OutlineTransparency = 0
                        highlight.Parent = char
                        esp.Highlight = highlight
                    end
                    esp.Highlight.Enabled = true
                else
                    if esp.Highlight then
                        esp.Highlight.Enabled = false
                    end
                end
            else
                esp.Box.Visible = false
                for _, line in pairs(esp.Skeleton) do line.Visible = false end
                esp.HealthBG.Visible = false
                esp.HealthBar.Visible = false
            end
        else
            esp.Box.Visible = false
            for _, line in pairs(esp.Skeleton) do line.Visible = false end
            esp.HealthBG.Visible = false
            esp.HealthBar.Visible = false
            if esp.Highlight then esp.Highlight.Enabled = false end
        end
    end

    -- ===== RAGE FUNCTIONS =====

    -- Third Person
    if settings.ThirdPerson then
        local char = LocalPlayer.Character
        local root = char and GetRootPart(char)
        if root then
            local offset = Vector3.new(0, 3, -8)
            local targetCFrame = CFrame.new(root.Position) * CFrame.Angles(0, 0, 0) * CFrame.new(offset)
            Camera.CFrame = targetCFrame
        end
    end

    -- No Recoil (восстановление камеры)
    if settings.NoRecoil then
        -- Простое решение: если камера дёрнулась, вернуть её в предыдущее положение
        -- (здесь можно добавить логику, но в общем случае это работает через сброс CFrame)
        -- Вместо этого мы просто игнорируем отдачу, блокируя RemoteEvent (если найдём)
        -- Пока заглушка: ничего не делаем, но можно добавить hook для отдачи
    end

    -- Spinbot
    if settings.Spinbot then
        local char = LocalPlayer.Character
        local root = char and GetRootPart(char)
        if root then
            root.CFrame = root.CFrame * CFrame.Angles(0, settings.SpinSpeed * dt, 0)
        end
    end

    -- Обычный аим (если включён и не Third Person)
    if settings.AimEnabled and not settings.SilentAim and not settings.ThirdPerson then
        local target = GetClosestTarget()
        if target then
            local targetCFrame = CFrame.lookAt(Camera.CFrame.Position, target.Position)
            local alpha = 1 - math.exp(-settings.Smoothness * dt * 60)
            Camera.CFrame = Camera.CFrame:Lerp(targetCFrame, alpha)
        end
    end

    -- Fake Silent Aim (если выбран или hook не работает)
    if settings.SilentAim and (settings.SilentMethod == "Fake" or not hookWorks) then
        if UserInputService:IsMouseButtonPressed(Enum.UserInputType.MouseButton1) then
            local target = GetClosestTarget()
            if target then
                local oldCFrame = Camera.CFrame
                Camera.CFrame = CFrame.lookAt(Camera.CFrame.Position, target.Position)
                task.wait(0.01)
                Camera.CFrame = oldCFrame
            end
        end
    end
end)
